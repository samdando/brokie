create or replace function public.is_crypto(p_market public.market) returns boolean
language sql immutable as $$ select p_market::text in ('btc','eth','sol') $$;
create or replace function public.weekend_mode(p_at timestamptz default now()) returns boolean
language sql stable as $$
 select extract(isodow from p_at at time zone 'America/New_York')=6
 or (extract(isodow from p_at at time zone 'America/New_York')=5 and (p_at at time zone 'America/New_York')::time >= time '17:00')
 or (extract(isodow from p_at at time zone 'America/New_York')=7 and (p_at at time zone 'America/New_York')::time < time '18:00')
$$;
create or replace function public.market_session_open(p_market public.market,p_at timestamptz default now()) returns boolean
language sql stable as $$
 select case when is_crypto(p_market) then weekend_mode(p_at) else
 not weekend_mode(p_at) and ((p_at at time zone 'America/New_York')::time < time '17:00'
 or (p_at at time zone 'America/New_York')::time >= time '18:00') end
$$;
-- Brokie's flat-at-close rule: weekdays at 17:00 NY; weekend crypto at Sunday 18:00 NY.
-- Uses the named timezone rather than UTC offsets so DST is automatic.
create or replace function public.position_session_end(p_market public.market,p_opened timestamptz) returns timestamptz
language plpgsql stable as $$
declare local_open timestamp := p_opened at time zone 'America/New_York'; d date := local_open::date;
begin
 if is_crypto(p_market) then
   d := d + ((7-extract(isodow from d)::int)%7);
   if d + time '18:00' <= local_open then d:=d+7; end if;
   return (d+time '18:00') at time zone 'America/New_York';
 end if;
 if local_open::time >= time '17:00' then d:=d+1; end if;
 while extract(isodow from d) in (6,7) loop d:=d+1; end loop;
 return (d+time '17:00') at time zone 'America/New_York';
end $$;
alter table public.positions add column if not exists session_closes_at timestamptz;
update public.positions set session_closes_at=public.position_session_end(market,opened_at) where session_closes_at is null;
create index if not exists positions_session_close on public.positions(session_closes_at);

-- An authenticated client cannot pick the session time or bypass the calendar.
create or replace function public.set_position_session() returns trigger
language plpgsql security definer set search_path=public as $$
begin
 if not market_session_open(new.market,now()) then raise exception 'This market session is closed.'; end if;
 if exists(select 1 from positions where user_id=new.user_id and session_closes_at<=now()) then
   raise exception 'A previous session is awaiting settlement. Please try again shortly.';
 end if;
 new.session_closes_at:=position_session_end(new.market,new.opened_at);
 return new;
end $$;
drop trigger if exists set_position_session on public.positions;
create trigger set_position_session before insert on public.positions for each row execute function public.set_position_session();

-- Runs without quotes arriving or clients connected. Never uses a reopening price.
-- If no reliable closing-session mark exists, leave the position pending for investigation.
create or replace function public.settle_market_closes(p_now timestamptz default now()) returns jsonb
language plpgsql security definer set search_path=public as $$
declare a accounts; p positions; mark numeric; quote_time timestamptz; value numeric; n int:=0; pending int:=0;
begin
 for a in select * from accounts x where exists(select 1 from positions o where o.user_id=x.user_id
   and o.season=x.season and o.session_closes_at<=p_now) order by user_id,season for update skip locked loop
   for p in select * from positions where user_id=a.user_id and season=a.season and session_closes_at<=p_now
     order by session_closes_at,id for update loop
     mark:=null; quote_time:=null;
     select x.price,x.at into mark,quote_time from (
       select q.price,q.quoted_at as at from quotes q where q.market=p.market
       union all select t.price,t.quoted_at from quote_ticks t where t.market=p.market
         and t.quoted_at>=p.session_closes_at-interval '5 minutes' and t.quoted_at<=p.session_closes_at
       union all select b.close,b.last_at from bars b where b.market=p.market and b.secs=60
         and b.idx between floor(extract(epoch from p.session_closes_at-interval '5 minutes')/60)::bigint
                       and floor(extract(epoch from p.session_closes_at)/60)::bigint
     ) x where x.at<=p.session_closes_at and x.at>=p.session_closes_at-interval '5 minutes'
       and x.at>=p.opened_at order by x.at desc limit 1;
     if mark is null then pending:=pending+1; continue; end if;
     select (mark-p.entry_price)*m.point_value*p.quantity*direction(p.side) into value from markets m where m.id=p.market;
     insert into trades(id,user_id,season,market,side,quantity,entry_price,exit_price,pnl,opened_at,closed_at,reason,exit_quoted_at)
       values(p.id,p.user_id,p.season,p.market,p.side,p.quantity,p.entry_price,mark,value,p.opened_at,p.session_closes_at,'market_close',quote_time);
     update accounts set cash=cash+value where user_id=p.user_id and season=p.season;
     delete from positions where id=p.id;
     n:=n+1;
   end loop;
   if not exists(select 1 from positions where user_id=a.user_id and season=a.season and session_closes_at<=p_now) then
     perform enforce_rules(a.user_id,a.season);
   end if;
 end loop;
 return jsonb_build_object('settled',n,'pending',pending);
end $$;
revoke all on function public.settle_market_closes(timestamptz),public.set_position_session() from public,anon,authenticated;

-- Prevent a manual close after the deadline from substituting a later/reopening mark.
-- Automatic settlement removes the position; manual attempts can retry after the next job.
create or replace function public.guard_session_exit() returns trigger
language plpgsql security definer set search_path=public as $$
begin
 if new.reason <> 'market_close' and exists(select 1 from positions p where p.user_id=new.user_id
   and p.market=new.market and p.side=new.side and p.opened_at=new.opened_at and p.session_closes_at<=now()) then
   raise exception 'Session ended. This position is awaiting its market-close receipt.';
 end if;
 return new;
end $$;
drop trigger if exists guard_session_exit on public.trades;
create trigger guard_session_exit before insert on public.trades for each row execute function public.guard_session_exit();
revoke all on function public.guard_session_exit() from public,anon,authenticated;

create or replace function public.receipt_day(p_closed timestamptz,p_reason text) returns date
language sql stable as $$ select trading_day_of(p_closed-case when p_reason='market_close' then interval '1 microsecond' else interval '0' end) $$;
create index if not exists trades_daily on public.trades(closed_at,user_id);
create or replace function public.daily_leaderboard(p_day date default null,p_squad uuid default null) returns jsonb
language plpgsql security definer set search_path=public as $$
declare uid uuid:=require_user(); d date:=coalesce(p_day,trading_today()); result jsonb;
begin
 if p_squad is not null then perform require_member(p_squad); end if;
 if d>trading_today() or d<trading_today()-365 then raise exception 'Choose a date within the past year.'; end if;
 select coalesce(jsonb_agg(to_jsonb(r) order by r.pnl desc,r.handle),'[]'::jsonb) into result from (
   select p.handle,p.id=uid as is_you,coalesce(round(sum(t.pnl),2),0) as pnl,
     count(t.id) as trades,count(t.id) filter(where t.pnl>0) as wins,count(t.id) filter(where t.pnl<0) as losses,
     count(t.id) filter(where t.pnl=0) as breakevens
   from profiles p left join trades t on t.user_id=p.id and receipt_day(t.closed_at,t.reason)=d
   where case when p_squad is not null then exists(select 1 from squad_members m where m.squad_id=p_squad and m.user_id=p.id)
     else (p.show_global or p.id=uid) and exists(select 1 from accounts a where a.user_id=p.id) end
   group by p.id,p.handle
 ) r;
 return jsonb_build_object('day',d,'today',trading_today(),'rows',result,'timezone','America/New_York','reset_hour',17);
end $$;

create or replace function public.daily_receipts(p_handle text,p_day date,p_squad uuid default null) returns jsonb
language plpgsql security definer set search_path=public as $$
declare uid uuid:=require_user(); owner uuid; visible boolean;
begin
 select id,show_global into owner,visible from profiles where handle=lower(p_handle);
 if owner is null then raise exception 'Player not found.'; end if;
 if p_squad is not null then
   perform require_member(p_squad);
   if not exists(select 1 from squad_members where squad_id=p_squad and user_id=owner) then raise exception 'Player is not in this squad.'; end if;
 elsif owner<>uid and not coalesce(visible,false) then raise exception 'This player is private.'; end if;
 if p_day>trading_today() or p_day<trading_today()-365 then raise exception 'Choose a date within the past year.'; end if;
 return coalesce((select jsonb_agg(to_jsonb(t) order by t.closed_at desc) from (
   select id,market,side,quantity,entry_price,exit_price,pnl,opened_at,closed_at,reason
   from trades where user_id=owner and receipt_day(closed_at,reason)=p_day
 ) t),'[]'::jsonb);
end $$;

create or replace function public.review_trade(p_trade uuid) returns jsonb
language plpgsql security definer set search_path=public as $$
declare uid uuid:=require_user(); t trades; size int; data jsonb;
begin
 select * into t from trades where id=p_trade and user_id=uid;
 if not found then raise exception 'Trade not found.'; end if;
 size:=case when t.closed_at-t.opened_at>interval '7 days' then 86400 when t.closed_at-t.opened_at>interval '6 hours' then 3600 else 60 end;
 select coalesce(jsonb_agg(jsonb_build_object('index',b.idx,'open',b.open,'high',b.high,'low',b.low,'close',b.close) order by b.idx),'[]'::jsonb)
 into data from bars b where b.market=t.market and b.secs=size
   and b.idx between floor(extract(epoch from t.opened_at)/size)::bigint-3 and floor(extract(epoch from t.closed_at)/size)::bigint+3;
 return jsonb_build_object('trade',to_jsonb(t),'candles',data,'bucket_seconds',size,
   'note',coalesce((select note from trade_notes where trade_id=t.id),''));
end $$;

create or replace function public.enforce_rules(p_user uuid, p_season date) returns text
language plpgsql security definer set search_path = public as $$
declare
  a accounts;
  eq numeric;
  today date := trading_today();
begin
  select * into a from accounts where user_id = p_user and season = p_season for update;
  if not found then return null; end if;
  -- Freeze risk checks until overdue positions have an authentic closing receipt.
  if exists(select 1 from positions where user_id=p_user and season=p_season and session_closes_at<=now()) then return a.status; end if;
  eq := a.cash + unrealized(p_user, p_season);

  -- New session (5pm New York). Equity at the rollover is yesterday's closing balance:
  -- it can raise the end-of-day high, then gets a fresh daily-loss allowance. A locked account unlocks.
  if a.trading_day is distinct from today then
    update accounts set trading_day = today, day_start_equity = eq,
                        eod_peak = greatest(coalesce(eod_peak, start_balance), eq),
                        status = case when status = 'locked' then 'active' else status end
     where user_id = p_user and season = p_season returning * into a;
  end if;
  if a.status = 'blown' then return 'blown'; end if;

  -- The intraday peak is still tracked for stats and for the trailing hard mode.
  if eq > a.peak_equity then
    update accounts set peak_equity = eq where user_id = p_user and season = p_season returning * into a;
  end if;
  update accounts set worst_drawdown = greatest(worst_drawdown, round((1 - eq / nullif(a.peak_equity, 0)) * 100, 4))
   where user_id = p_user and season = p_season;

  -- Max loss limit -> blown for the season.
  if eq <= dd_floor(a) then
    perform close_all(p_user, p_season, 'blown');
    update accounts set status = 'blown', blown_at = now(),
                        protection_credit = protection_credit + greatest(0, -cash), cash = greatest(0, cash)
     where user_id = p_user and season = p_season;
    return 'blown';
  end if;

  -- Daily loss -> locked until the next session.
  if a.status = 'active' and eq <= a.day_start_equity * (1 - a.daily_loss_pct / 100) then
    perform close_all(p_user, p_season, 'daily_limit');
    update accounts set status = 'locked', locked_at = now() where user_id = p_user and season = p_season;
    return 'locked';
  end if;

  -- Profit target -> 🎯 target hit (sticky). Funded needs the target AND 5 trading days, checked
  -- together: equity has to be at +10% on a check after the fifth trading day. Sticky for the season.
  if eq >= a.start_balance * 1.10 then
    if a.target_hit_at is null then
      update accounts set target_hit_at = now() where user_id = p_user and season = p_season;
    end if;
    if a.funded_at is null and trading_days(p_user, p_season) >= 5 then
      update accounts set funded_at = now() where user_id = p_user and season = p_season;
    end if;
  end if;
  return a.status;
end $$;
create or replace function public.ingest_quotes(p_quotes jsonb) returns jsonb
language plpgsql security definer set search_path = public as $$
declare
  item jsonb;
  m markets;
  px numeric;
  at timestamptz;
  accepted int := 0;
  breached int := 0;
  acct record;
begin
  for item in select * from jsonb_array_elements(coalesce(p_quotes, '[]'::jsonb)) loop
    m := null;
    select * into m from markets where id::text = item->>'market';
    px := (item->>'price')::numeric;
    at := coalesce((item->>'quoted_at')::timestamptz, now());
    continue when m.id is null or px is null or px <= 0 or px > 10000000 or at > now() + interval '5 seconds';
    px := round(px / m.tick_size) * m.tick_size;
    insert into quote_ticks (market, price, quoted_at) values (m.id, px, at);
    insert into quotes (market, price, source, quoted_at, updated_at)
    values (m.id, px, coalesce(item->>'source', 'feed'), at, now())
    on conflict (market) do update
      set price = excluded.price, source = excluded.source, quoted_at = excluded.quoted_at, updated_at = now()
      where quotes.quoted_at <= excluded.quoted_at;
    accepted := accepted + 1;
  end loop;

  perform settle_market_closes();
  for acct in select distinct user_id, season from positions where season = current_season() loop
    if enforce_rules(acct.user_id, acct.season) <> 'active' then breached := breached + 1; end if;
  end loop;
  perform settle_old_seasons(null);
  delete from quote_ticks where quoted_at < now() - interval '3 days';
  return jsonb_build_object('accepted', accepted, 'liquidated', breached);
end $$;
create or replace function public.settle_old_seasons(p_user uuid default null) returns int
language plpgsql security definer set search_path=public as $$
declare a accounts; pos positions; n int:=0;
begin
 for a in select * from accounts x where x.season<current_season() and (p_user is null or x.user_id=p_user)
   and exists(select 1 from positions o where o.user_id=x.user_id and o.season=x.season)
   order by user_id,season for update skip locked loop
  for pos in select * from positions where user_id=a.user_id and season=a.season
   and (session_closes_at is null or session_closes_at>now()) for update loop
   perform settle_position(pos,'season_end'); n:=n+1;
  end loop;
 end loop;
 return n;
end $$;
create or replace function public.squad_leaderboard(p_squad uuid) returns jsonb
language plpgsql security definer set search_path = public as $$
declare uid uuid := require_member(p_squad);
begin
  return coalesce((
    select jsonb_agg(row_to_json(r) order by r.qualified desc, r.equity desc, r.handle)
    from (
      select p.handle,
             (select coalesce(round(sum(t.pnl),2),0) from trades t where t.user_id=p.id and receipt_day(t.closed_at,t.reason)=trading_today()) as today_pnl,
             p.id = uid as is_you,
             round(coalesce(a.cash, starting_balance()) + unrealized(p.id, current_season()), 2) as equity,
             round((coalesce(a.cash, starting_balance()) + unrealized(p.id, current_season()) - coalesce(a.start_balance, starting_balance()))
                   / coalesce(a.start_balance, starting_balance()) * 100, 2) as return_pct,
             (select count(*) from trades t where t.user_id = p.id and t.season = current_season()) as trades,
             (select coalesce(min(t.pnl), 0) from trades t where t.user_id = p.id and t.season = current_season()) as worst_trade,
             (select count(*) from trades t where t.user_id = p.id and t.season = current_season() and t.pnl > 0) as wins,
             (select count(*) from trades t where t.user_id = p.id and t.season = current_season() and t.pnl < 0) as losses,
             (select count(*) from positions o where o.user_id = p.id and o.season = current_season()) as open_positions,
             coalesce(a.status, 'active') as status,
             a.funded_at is not null as funded,
             a.target_hit_at is not null as target_hit,
             coalesce(a.resets_used, 0) as resets_used,
             trading_days(p.id, current_season()) as trading_days,
             trading_days(p.id, current_season()) >= 5 as qualified,
             coalesce(a.worst_drawdown, 0) as worst_drawdown
      from squad_members m
      join profiles p on p.id = m.user_id
      left join accounts a on a.user_id = p.id and a.season = current_season()
      where m.squad_id = p_squad
    ) r), '[]'::jsonb);
end $$;
create or replace function public.global_leaderboard(p_limit int default 50) returns jsonb
language plpgsql security definer set search_path = public as $$
declare
  uid uuid := require_user();
  lim int := greatest(5, least(coalesce(p_limit, 50), 100));
  rows jsonb;
  me jsonb;
  total int;
begin
  create temp table _gl on commit drop as
  with base as (
    select p.id, p.handle, p.show_global,
           (select coalesce(round(sum(t.pnl),2),0) from trades t where t.user_id=p.id and receipt_day(t.closed_at,t.reason)=trading_today()) as today_pnl,
           round(coalesce(a.cash, starting_balance()) + unrealized(p.id, current_season()), 2) as equity,
           coalesce(a.start_balance, starting_balance()) as start_balance,
           coalesce(a.status, 'active') as status,
           a.funded_at is not null as funded,
           trading_days(p.id, current_season()) as trading_days,
           (select count(*) from trades t where t.user_id = p.id and t.season = current_season()) as trades
    from profiles p
    join accounts a on a.user_id = p.id and a.season = current_season()
  ), scored as (
    select b.*, round((b.equity - b.start_balance) / b.start_balance * 100, 2) as return_pct,
           b.trading_days >= 5 as qualified
    from base b
  )
  select s.*, row_number() over (order by s.qualified desc, s.return_pct desc, s.handle) as rank from scored s;

  select count(*) into total from _gl where show_global or id = uid;

  select coalesce(jsonb_agg(to_jsonb(r) order by r.rank), '[]'::jsonb) into rows from (
    select g.rank, g.handle, g.id = uid as is_you, g.equity, g.return_pct, g.status, g.funded,
           g.qualified, g.trading_days, g.trades, g.today_pnl
    from _gl g where g.show_global or g.id = uid
    order by g.rank limit lim
  ) r;

  select to_jsonb(r) into me from (
    select g.rank, g.handle, true as is_you, g.equity, g.return_pct, g.status, g.funded,
           g.qualified, g.trading_days, g.trades, g.today_pnl, g.show_global
    from _gl g where g.id = uid
  ) r;

  drop table _gl;
  return jsonb_build_object('rows', rows, 'me', me, 'players', total, 'season', current_season());
end $$;
commit;

create or replace function public.position_mark(p positions) returns numeric language sql stable security definer set search_path=public as $$
 select case when p.market='prediction' then
  (select case when m.payout_yes is not null then case when p.prediction_outcome='yes' then m.payout_yes else 1-m.payout_yes end
   else case when p.prediction_outcome='yes' then m.yes_bid else m.no_bid end end from prediction_markets m where m.id=p.prediction_id)
 else (select price from quotes where market=p.market) end
$$;
create or replace function public.unrealized(p_user uuid,p_season date) returns numeric language sql stable security definer set search_path=public as $$
 select coalesce(sum((coalesce(position_mark(p),p.entry_price)-p.entry_price)*m.point_value*p.quantity*direction(p.side)),0)
 from positions p join markets m on m.id=p.market where p.user_id=p_user and p.season=p_season
$$;
create or replace function public.position_margin(p positions) returns numeric language sql immutable as $$
 select case when p.market='prediction' then p.entry_price*p.quantity else p.quantity*1000 end
$$;
create or replace function public.set_position_session() returns trigger language plpgsql security definer set search_path=public as $$
begin
 if new.market='prediction' then new.session_closes_at:=season_ends_at(new.season);
 else
  if not market_session_open(new.market,now()) then raise exception 'This market session is closed.'; end if;
  new.session_closes_at:=position_session_end(new.market,new.opened_at);
 end if;
 if exists(select 1 from positions where user_id=new.user_id and session_closes_at<=now()) then raise exception 'A previous session is awaiting settlement.'; end if;
 return new;
end $$;
create or replace function public.settle_position(p_position positions,p_reason text) returns trades
language plpgsql security definer set search_path=public as $$
declare mark numeric; value numeric; result trades;
begin
 mark:=position_mark(p_position);
 if mark is null then raise exception 'No recorded price available.'; end if;
 select (mark-p_position.entry_price)*m.point_value*p_position.quantity*direction(p_position.side) into value from markets m where m.id=p_position.market;
 update accounts set cash=cash+value where user_id=p_position.user_id and season=p_position.season;
 insert into trades(id,user_id,season,market,side,quantity,entry_price,exit_price,pnl,opened_at,reason,prediction_id,prediction_outcome,prediction_question,exit_quoted_at)
 values(p_position.id,p_position.user_id,p_position.season,p_position.market,p_position.side,p_position.quantity,p_position.entry_price,mark,value,p_position.opened_at,p_reason,
 p_position.prediction_id,p_position.prediction_outcome,(select title from prediction_markets where id=p_position.prediction_id),
 case when p_position.market='prediction' then (select quoted_at from prediction_markets where id=p_position.prediction_id) else (select quoted_at from quotes where market=p_position.market) end)
 returning * into result;
 delete from positions where id=p_position.id;
 return result;
end $$;

create or replace function public.open_prediction(p_market text,p_outcome text,p_stake numeric default 100) returns jsonb
language plpgsql security definer set search_path=public as $$
declare uid uuid:=require_user(); a accounts; m prediction_markets; ask numeric; qty int; used numeric; p positions;
begin
 if p_outcome is null or p_outcome not in ('yes','no') or p_stake is null or p_stake<1 or p_stake>1000 then raise exception 'Choose Yes or No and a paper stake between $1 and $1,000.'; end if;
 a:=ensure_account(uid);
 select * into a from accounts where user_id=uid and season=a.season for update;
 if a.status<>'active' then raise exception 'This account is locked or blown.'; end if;
 select * into m from prediction_markets where id=p_market;
 if not found or m.status<>'active' or m.close_at<=now() or m.quoted_at<now()-interval '90 seconds' or m.quoted_at>now()+interval '5 seconds' or m.payout_yes is not null then raise exception 'This prediction is closed or its prices are stale.'; end if;
 ask:=case when p_outcome='yes' then m.yes_ask else m.no_ask end;
 if ask<=0 or ask>=1 then raise exception 'No executable price for this outcome.'; end if;
 qty:=least(10000,floor(p_stake/ask)::int);
 if qty<1 then raise exception 'Choose a larger paper stake.'; end if;
 select coalesce(sum(position_margin(o)),0) into used from positions o where user_id=uid and season=a.season;
 if a.cash+unrealized(uid,a.season)-used<qty*ask then raise exception 'Not enough buying power.'; end if;
 if (select count(*) from positions where user_id=uid and season=a.season)>=10 then raise exception 'Max 10 open positions.'; end if;
 if (select count(*) from positions where user_id=uid and opened_at>now()-interval '10 seconds')>=5 then raise exception 'Slow down — max 5 new trades per 10 seconds.'; end if;
 insert into positions(user_id,season,market,side,quantity,entry_price,prediction_id,prediction_outcome)
 values(uid,a.season,'prediction','long',qty,ask,m.id,p_outcome) returning * into p;
 perform enforce_rules(uid,a.season);
 return to_jsonb(p);
end $$;
create or replace function public.close_prediction(p_id uuid) returns jsonb language plpgsql security definer set search_path=public as $$
declare uid uuid:=require_user(); p positions; m prediction_markets; t trades;
begin
 perform 1 from accounts where user_id=uid order by season for update;
 select * into p from positions where id=p_id and user_id=uid and market='prediction' for update;
 if not found then raise exception 'Prediction already closed.'; end if;
 select * into m from prediction_markets where id=p.prediction_id;
 if m.payout_yes is null and (m.status<>'active' or m.close_at<=now()) then raise exception 'This event has closed and is awaiting its official result.'; end if;
 if m.payout_yes is null and (m.quoted_at<now()-interval '90 seconds' or position_mark(p)<=0) then raise exception 'No fresh exit bid. Try again shortly.'; end if;
 t:=settle_position(p,case when m.payout_yes is not null then 'prediction_settled' else 'closed' end);
 perform enforce_rules(uid,p.season); return to_jsonb(t);
end $$;
-- Service-only ingestion validates all prices and settles only official final payouts.
create or replace function public.ingest_predictions(p_rows jsonb) returns jsonb language plpgsql security definer set search_path=public as $$
declare r jsonb; a accounts; p positions; n int:=0; settled int:=0;
begin
 for r in select * from jsonb_array_elements(p_rows) loop
  if (r->>'quoted_at')::timestamptz<now()-interval '90 seconds' or (r->>'quoted_at')::timestamptz>now()+interval '5 seconds' then continue; end if;
  insert into prediction_markets(id,title,rules,close_at,status,yes_bid,yes_ask,no_bid,no_ask,payout_yes,quoted_at,volume)
  values(r->>'id',r->>'title',r->>'rules',(r->>'close_at')::timestamptz,r->>'status',(r->>'yes_bid')::numeric,(r->>'yes_ask')::numeric,
  (r->>'no_bid')::numeric,(r->>'no_ask')::numeric,(r->>'payout_yes')::numeric,(r->>'quoted_at')::timestamptz,coalesce((r->>'volume')::numeric,0))
  on conflict(id) do update set title=excluded.title,rules=excluded.rules,close_at=excluded.close_at,status=case when prediction_markets.payout_yes is not null then prediction_markets.status else excluded.status end,
  yes_bid=case when excluded.status='active' then excluded.yes_bid else prediction_markets.yes_bid end,
  yes_ask=case when excluded.status='active' then excluded.yes_ask else prediction_markets.yes_ask end,
  no_bid=case when excluded.status='active' then excluded.no_bid else prediction_markets.no_bid end,
  no_ask=case when excluded.status='active' then excluded.no_ask else prediction_markets.no_ask end,
  payout_yes=coalesce(prediction_markets.payout_yes,excluded.payout_yes),quoted_at=excluded.quoted_at,volume=excluded.volume
  where prediction_markets.quoted_at<=excluded.quoted_at;
  insert into prediction_ticks(id,quoted_at,yes_bid,no_bid) select id,quoted_at,
   coalesce(payout_yes,yes_bid),case when payout_yes is not null then 1-payout_yes else no_bid end
   from prediction_markets where id=r->>'id' on conflict do nothing;
  n:=n+1;
 end loop;
 for a in select * from accounts x where exists(select 1 from positions o where o.user_id=x.user_id and o.season=x.season and o.market='prediction') order by user_id,season for update skip locked loop
  for p in select o.* from positions o join prediction_markets m on m.id=o.prediction_id where o.user_id=a.user_id and o.season=a.season and m.payout_yes is not null for update of o loop
   perform settle_position(p,'prediction_settled'); settled:=settled+1;
  end loop;
  perform enforce_rules(a.user_id,a.season);
 end loop;
 delete from prediction_ticks where quoted_at<now()-interval '7 days';
 return jsonb_build_object('accepted',n,'settled',settled);
end $$;

$$;

create or replace function public.open_position(p_market public.market, p_side public.trade_side, p_quantity int)
returns jsonb
language plpgsql security definer set search_path = public as $$
declare
  uid uuid := require_user();
  acct accounts;
  q quotes;
  equity numeric;
  used numeric;
  pos positions;
  recent int;
begin
  if p_quantity not between 1 and 10 then
    raise exception 'Choose 1 to 10 contracts.' using errcode = 'P0001';
  end if;
  acct := ensure_account(uid);
  select * into acct from accounts where user_id = uid and season = acct.season for update;
  if acct.status = 'blown' then
    raise exception 'Account blown for this season. 💀 Use your buy-back or wait for next season.' using errcode = 'P0001';
  end if;
  if acct.status = 'locked' then
    raise exception 'Daily loss limit hit. Trading unlocks when the next session opens (5pm New York, % Brisbane).',
      to_char(trading_day_ends_at() at time zone 'Australia/Brisbane', 'FMHH12am') using errcode = 'P0001';
  end if;
  select count(*) into recent from positions where user_id = uid and opened_at > now() - interval '10 seconds';
  if recent >= 5 then raise exception 'Slow down — max 5 new trades per 10 seconds.' using errcode = 'P0001'; end if;
  select * into q from quotes where market = p_market;
  if not found or q.quoted_at < now() - quote_max_age() then
    raise exception 'Market is closed or the price feed is stale. Try again when it''s live.' using errcode = 'P0001';
  end if;
  if (select count(*) from positions where user_id = uid and season = acct.season) >= 10 then
    raise exception 'Max 10 open positions. Close one first.' using errcode = 'P0001';
  end if;
  equity := acct.cash + unrealized(uid, acct.season);
  select coalesce(sum(position_margin(positions)), 0) into used from positions where user_id = uid and season = acct.season;
  if equity - used + 0.000001 < p_quantity * 1000 then
    raise exception 'Not enough buying power. Close a position or choose fewer contracts.' using errcode = 'P0001';
  end if;
  insert into positions (user_id, season, market, side, quantity, entry_price)
  values (uid, acct.season, p_market, p_side, p_quantity, q.price)
  returning * into pos;
  return to_jsonb(pos);
end $$;

create or replace function public.reduce_position(p_market public.market, p_side public.trade_side, p_quantity int) returns jsonb
language plpgsql security definer set search_path = public as $$
declare
  uid uuid := require_user();
  v_season date := current_season();
  left_to_close int := p_quantity;
  leg positions;
  part positions;
  t trades;
  total numeric := 0;
  closed int := 0;
begin
  if p_market='prediction' then raise exception 'Use the prediction exit button.'; end if;
  if p_quantity is null or p_quantity < 1 then
    raise exception 'Choose at least 1 micro to close.' using errcode = 'P0001';
  end if;
  perform 1 from accounts where user_id = uid and accounts.season = v_season for update;
  if not exists (select 1 from positions where user_id = uid and positions.season = v_season and market = p_market and side = p_side) then
    raise exception 'No % position on that market to take a partial from.', p_side using errcode = 'P0001';
  end if;

  for leg in
    select * from positions
     where user_id = uid and positions.season = v_season and market = p_market and side = p_side
     order by opened_at desc, id desc
     for update
  loop
    exit when left_to_close <= 0;
    if leg.quantity <= left_to_close then
      t := settle_position(leg, 'closed');
      left_to_close := left_to_close - leg.quantity;
      closed := closed + leg.quantity;
    else
      -- Split: settle the closed slice as its own trade (new id), keep the rest open.
      part := leg;
      part.id := gen_random_uuid();
      part.quantity := left_to_close;
      t := settle_position(part, 'closed');   -- its delete finds no row; only the trade + cash move
      update positions set quantity = quantity - left_to_close where id = leg.id;
      closed := closed + left_to_close;
      left_to_close := 0;
    end if;
    total := total + t.pnl;
  end loop;

  perform enforce_rules(uid, v_season);
  return jsonb_build_object('closed', closed, 'pnl', round(total, 2));
end $$;

create or replace function public.settle_market_closes(p_now timestamptz default now()) returns jsonb
language plpgsql security definer set search_path=public as $$
declare a accounts; p positions; mark numeric; quote_time timestamptz; value numeric; n int:=0; pending int:=0;
begin
 for a in select * from accounts x where exists(select 1 from positions o where o.user_id=x.user_id
   and o.season=x.season and o.session_closes_at<=p_now) order by user_id,season for update skip locked loop
   for p in select * from positions where user_id=a.user_id and season=a.season and session_closes_at<=p_now
     order by session_closes_at,id for update loop
     mark:=null; quote_time:=null;
     if p.market='prediction' then
       select case when p.prediction_outcome='yes' then t.yes_bid else t.no_bid end,t.quoted_at into mark,quote_time
       from prediction_ticks t where t.id=p.prediction_id and t.quoted_at between p.session_closes_at-interval '5 minutes' and p.session_closes_at
       order by t.quoted_at desc limit 1;
     else
     select x.price,x.at into mark,quote_time from (
       select q.price,q.quoted_at as at from quotes q where q.market=p.market
       union all select t.price,t.quoted_at from quote_ticks t where t.market=p.market
         and t.quoted_at>=p.session_closes_at-interval '5 minutes' and t.quoted_at<=p.session_closes_at
       union all select b.close,b.last_at from bars b where b.market=p.market and b.secs=60
         and b.idx between floor(extract(epoch from p.session_closes_at-interval '5 minutes')/60)::bigint
                       and floor(extract(epoch from p.session_closes_at)/60)::bigint
     ) x where x.at<=p.session_closes_at and x.at>=p.session_closes_at-interval '5 minutes'
       and x.at>=p.opened_at order by x.at desc limit 1;
     end if;
     if mark is null then pending:=pending+1; continue; end if;
     select (mark-p.entry_price)*m.point_value*p.quantity*direction(p.side) into value from markets m where m.id=p.market;
     insert into trades(id,user_id,season,market,side,quantity,entry_price,exit_price,pnl,opened_at,closed_at,reason,exit_quoted_at,prediction_id,prediction_outcome,prediction_question)
       values(p.id,p.user_id,p.season,p.market,p.side,p.quantity,p.entry_price,mark,value,p.opened_at,p.session_closes_at,'market_close',quote_time,p.prediction_id,p.prediction_outcome,(select title from prediction_markets where id=p.prediction_id));
     update accounts set cash=cash+value where user_id=p.user_id and season=p.season;
     delete from positions where id=p.id;
     n:=n+1;
   end loop;
   if not exists(select 1 from positions where user_id=a.user_id and season=a.season and session_closes_at<=p_now) then
     perform enforce_rules(a.user_id,a.season);
   end if;
 end loop;
 return jsonb_build_object('settled',n,'pending',pending);
end $$;

create or replace function public.chart_bubbles(p_market public.market, p_squad uuid default null) returns jsonb
language plpgsql security definer set search_path = public as $$
declare
  uid uuid := require_user();
begin
  if p_squad is not null then perform require_member(p_squad); end if;
  return coalesce((
    select jsonb_agg(row_to_json(b) order by b.at)
    from (
      select 'open'::text as state, o.id, p.handle, o.user_id = uid as is_you, o.side, o.quantity, o.entry_price,
             null::numeric as exit_price, null::numeric as pnl, null::text as reason, o.opened_at as at, o.opened_at,
             round(position_margin(o) / nullif(a.cash + unrealized(o.user_id, o.season), 0) * 100, 2) as size_pct,
             a.peak_equity * (1 - a.max_dd_pct / 100) as dd_floor,
             a.cash + unrealized(o.user_id, o.season) as equity, o.prediction_id, o.prediction_outcome,
             (select title from prediction_markets where id=o.prediction_id) as prediction_question, position_mark(o) as mark_price
      from positions o
      join profiles p on p.id = o.user_id
      join accounts a on a.user_id = o.user_id and a.season = o.season
      where o.market = p_market and o.season = current_season()
        and (o.user_id = uid or (p_squad is not null and exists (select 1 from squad_members m where m.squad_id = p_squad and m.user_id = o.user_id)))
      union all
      select * from (
        select 'closed'::text, t.id, p.handle, t.user_id = uid, t.side, t.quantity, t.entry_price, t.exit_price, t.pnl, t.reason,
               t.closed_at, t.opened_at, null::numeric, null::numeric, null::numeric,
               t.prediction_id, t.prediction_outcome, t.prediction_question, t.exit_price
        from trades t join profiles p on p.id = t.user_id
        where t.market = p_market and t.season = current_season() and t.closed_at > now() - interval '24 hours'
          and (t.user_id = uid or (p_squad is not null and exists (select 1 from squad_members m where m.squad_id = p_squad and m.user_id = t.user_id)))
        order by t.closed_at desc limit 60
      ) recent
    ) b), '[]'::jsonb);
end $$;

// Server-side Coinbase quotes for Brokie paper contracts. Called by our secret-protected cron.
// https://docs.cdp.coinbase.com/api-reference/exchange-api/rest-api/products/get-product-ticker
const products = { btc: "BTC-USD", eth: "ETH-USD", sol: "SOL-USD" };
export function parseTicker(market: string, payload: { price?: string; time?: string }, now = Date.now()) {
  const price = Number(payload.price), at = Date.parse(payload.time ?? "");
  if (!Object.hasOwn(products, market) || !Number.isFinite(price) || price <= 0 || price > 10_000_000 || !Number.isFinite(at) || now-at > 45_000 || at-now > 5000) return null;
  return { market, price, quoted_at: new Date(at).toISOString(), source: "coinbase" };
}
Deno.serve(async (req: Request) => {
  const key = Deno.env.get("BROKIE_SERVICE_KEY") ?? Deno.env.get("SUPABASE_SERVICE_ROLE_KEY");
  const base = Deno.env.get("SUPABASE_URL");
  if (!key || !base) return Response.json({ error: "Missing server configuration" }, { status: 500 });
  const headers: Record<string,string> = { apikey: key, "Content-Type": "application/json" };
  if (key.startsWith("eyJ")) headers.Authorization = `Bearer ${key}`;
  const rpc = (name: string, body: unknown) => fetch(`${base}/rest/v1/rpc/${name}`, { method: "POST", headers, body: JSON.stringify(body), signal: AbortSignal.timeout(8000) });
  const auth = await rpc("check_cron_secret", { p_secret: req.headers.get("x-cron-secret") ?? "" });
  if (!auth.ok || await auth.json() !== true) return new Response("Forbidden", { status: 403 });
  // Independent requests: a failure for one product never fabricates a quote or blocks the others.
  const results = await Promise.allSettled(Object.entries(products).map(async ([market, product]) => {
    const response = await fetch(`https://api.exchange.coinbase.com/products/${product}/ticker`, { headers: { Accept: "application/json" }, signal: AbortSignal.timeout(5000) });
    if (!response.ok) throw new Error(`${market}: HTTP ${response.status}`);
    return parseTicker(market, await response.json());
  }));
  const quotes = results.flatMap(r => r.status === "fulfilled" && r.value ? [r.value] : []);
  if (!quotes.length) return Response.json({ error: "No fresh crypto quotes" }, { status: 502 });
  const result = await rpc("ingest_quotes", { p_quotes: quotes });
  return new Response(await result.text(), { status: result.status, headers: { "Content-Type": "application/json" } });
});

// https://docs.kalshi.com/api-reference/market/get-markets
const FEED = "https://api.elections.kalshi.com/trade-api/v2";
export function parseMarket(m: Record<string, unknown>, at = new Date().toISOString()) {
  if (typeof m.ticker !== "string" || typeof m.title !== "string" || m.market_type !== "binary" || Number(m.notional_value_dollars) !== 1 || m.mve_collection_ticker) return null;
  const prices = [m.yes_bid_dollars,m.yes_ask_dollars,m.no_bid_dollars,m.no_ask_dollars].map(v => v == null || v === "" ? NaN : Number(v));
  if (prices.some(p => !Number.isFinite(p) || p < 0 || p > 1) || !Number.isFinite(Date.parse(String(m.close_time)))) return null;
  const [yes_bid,yes_ask,no_bid,no_ask] = prices;
  if (m.status === "active" && (yes_bid > yes_ask || no_bid > no_ask)) return null;
  // A tentative result is not a settlement. Respect final scalar/void payouts as well as Yes/No.
  let payout: number | null = null;
  if (m.status === "finalized" || m.status === "settled") {
    const value = m.settlement_value_dollars == null || m.settlement_value_dollars === "" ? NaN : Number(m.settlement_value_dollars);
    if (Number.isFinite(value) && value >= 0 && value <= 1) payout = value;
    else if (m.result === "yes") payout = 1;
    else if (m.result === "no") payout = 0;
  }
  const rules = [m.rules_primary,m.rules_secondary].filter(v => typeof v === "string" && v).join("\n\n");
  return { id:m.ticker,title:m.title,rules,close_at:m.close_time,status:rules ? m.status : "paused",yes_bid,yes_ask,no_bid,no_ask,payout_yes:payout,quoted_at:at,volume:Number(m.volume_24h_fp)||0 };
}
Deno.serve(async (req: Request) => {
  const key = Deno.env.get("BROKIE_SERVICE_KEY") ?? Deno.env.get("SUPABASE_SERVICE_ROLE_KEY"), base = Deno.env.get("SUPABASE_URL");
  if (!key || !base) return Response.json({ error:"Missing server configuration" },{status:500});
  const headers:Record<string,string>={apikey:key,"Content-Type":"application/json"};
  if (key.startsWith("eyJ")) headers.Authorization=`Bearer ${key}`;
  const rpc=(name:string,body:unknown)=>fetch(`${base}/rest/v1/rpc/${name}`,{method:"POST",headers,body:JSON.stringify(body),signal:AbortSignal.timeout(8000)});
  const check=await rpc("check_cron_secret",{p_secret:req.headers.get("x-cron-secret")??""});
  if(!check.ok || await check.json()!==true)return new Response("Forbidden",{status:403});
  const get=async(path:string)=>{const r=await fetch(`${FEED}${path}`,{headers:{Accept:"application/json"},signal:AbortSignal.timeout(6000)});if(!r.ok)throw new Error(`Feed ${r.status}`);return r.json();};
  const watchResponse=await rpc("prediction_watchlist",{});
  if(!watchResponse.ok)return Response.json({error:"Watchlist unavailable"},{status:502});
  const watch:string[]=await watchResponse.json();
  const requests=[...['KXBTC15M','KXNBAGAME'].map(series=>get(`/markets?status=open&limit=100&series_ticker=${series}`)),
    ...Array.from({length:Math.ceil(watch.length/50)},(_,i)=>get(`/markets?limit=100&tickers=${encodeURIComponent(watch.slice(i*50,i*50+50).join(','))}`))];
  const results=await Promise.allSettled(requests);
  const all=results.flatMap(r=>r.status==='fulfilled' && Array.isArray(r.value.markets)?r.value.markets:[]);
  const parsed=all.map(m=>parseMarket(m)).filter((m):m is NonNullable<ReturnType<typeof parseMarket>>=>m!==null);
  const tracked=parsed.filter(m=>watch.includes(m.id));
  const candidates=parsed.filter(m=>m.status==='active' && m.volume>0 && m.yes_ask-m.yes_bid<=0.2 && m.no_ask-m.no_bid<=0.2 && m.yes_ask>0 && m.yes_ask<1 && m.no_ask>0 && m.no_ask<1 && Date.parse(String(m.close_at))>Date.now()).sort((a,b)=>b.volume-a.volume).slice(0,12);
  const rows=[...new Map([...candidates,...tracked].map(m=>[m.id,m])).values()];
  if(!rows.length)return Response.json({error:"No fresh prediction markets"},{status:502});
  const out=await rpc("ingest_predictions",{p_rows:rows});
  return new Response(await out.text(),{status:out.status,headers:{"Content-Type":"application/json"}});
});
